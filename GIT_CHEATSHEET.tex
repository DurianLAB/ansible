\section{Basic Commands}
\begin{lstlisting}[style=bash]
# Check current status
git status

# Check differences
git diff

# Check staged differences
git diff --staged

# View commit history
git log --oneline

# View recent commits with details
git log -10 --oneline --graph --decorate
\end{lstlisting}

\section{Staging \& Committing}
\begin{lstlisting}[style=bash]
# Add specific file
git add file.txt

# Add all changes
git add .

# Commit with message
git commit -m "Your commit message"

# Add and commit in one command
git commit -am "Your commit message"

# Amend last commit (don't amend pushed commits!)
git commit --amend -m "New message"
\end{lstlisting}

\section{Git Diff (Viewing Differences)}
\begin{lstlisting}[style=bash]
# Show unstaged changes
git diff

# Show staged changes
git diff --staged

# Show changes between commits
git diff commit1 commit2

# Show changes in a specific file
git diff -- file.txt

# Show changes between branches
git diff main..feature-branch

# Show only filenames that changed
git diff --name-only

# Show diff stat (summary)
git diff --stat
\end{lstlisting}

\section{Git Log (Viewing History)}
\begin{lstlisting}[style=bash]
# Basic log view
git log

# Show last N commits
git log -5

# One line per commit
git log --oneline

# Show commits with graph
git log --graph --oneline --all

# Show commits by author
git log --author="John"

# Show commits in a date range
git log --since="2024-01-01"
git log --after="2 weeks ago"

# Show commits that touched specific files
git log -- path/to/file.txt
git log --follow path/to/file.txt

# Show commits that added/removed a string
git log -S "function_name"

# Show commits between two branches
git log main..feature-branch

# Show what changed in each commit
git log -p

# Show merge commits only
git log --merges
\end{lstlisting}

\section{Git Blame}
\begin{lstlisting}[style=bash]
# Show who changed what in a file
git blame file.txt

# Show blame with line numbers
git blame -L 10,20 file.txt

# Show blame ignoring whitespace
git blame -w file.txt
\end{lstlisting}

\section{Branching}
\begin{lstlisting}[style=bash]
# List branches
git branch

# List all branches (including remote)
git branch -a

# Create and switch to new branch
git checkout -b feature-branch
git switch -c feature-branch

# Switch branch
git checkout branch-name
git switch branch-name

# Switch to previous branch
git checkout -

# Delete local branch
git branch -d branch-name
git branch -D branch-name

# Delete remote branch
git push origin --delete branch-name

# Rename branch
git branch -m old-name new-name

# Set upstream branch
git branch -u origin/branch-name

# Track remote branch
git checkout --track origin/branch-name
\end{lstlisting}

\section{Branch Controlling \& Switching}
\begin{lstlisting}[style=bash]
# Switch to existing branch
git checkout main
git switch main

# Switch to previous branch
git checkout -
git switch -

# Create and switch to new branch
git checkout -b new-feature
git switch -c new-feature

# Track remote branch
git checkout -b local-name origin/remote-name

# List all branches with tracking info
git branch -vv

# Find common ancestor
git merge-base main feature-branch
\end{lstlisting}

\section{Cleaning Branches}
\begin{lstlisting}[style=bash]
# Delete local branch
git branch -d branch-name

# Delete remote branch
git push origin --delete branch-name

# Prune deleted remote branches
git remote prune origin
git fetch -p

# Delete branches merged into main
git checkout main
git branch --merged | grep -v "main" | xargs git branch -d

# Delete all local branches except main
git branch | grep -v "main" | xargs git branch -D
\end{lstlisting}

\section{Remote Operations}
\begin{lstlisting}[style=bash]
# Fetch updates from remote
git fetch

# Pull changes
git pull

# Pull with rebase
git pull --rebase

# Push to remote
git push

# Push new branch to remote
git push -u origin branch-name

# Force push (careful!)
git push -f origin branch-name

# Delete remote branch
git push origin --delete branch-name
\end{lstlisting}

\section{Stashing}
\begin{lstlisting}[style=bash]
# Stash current changes
git stash

# Stash with a message
git stash save "Work in progress"

# Stash including untracked files
git stash -u

# List all stashes
git stash list

# Show stash contents
git stash show
git stash show -p

# Apply latest stash (keep in stash)
git stash apply

# Apply and remove latest stash
git stash pop

# Drop latest stash
git stash drop

# Clear all stashes
git stash clear

# Create stash from specific files
git stash push -m "Message" -- path/to/file.txt
\end{lstlisting}

\section{Reflog (Reference Logs)}
\begin{lstlisting}[style=bash]
# Show reflog
git reflog
git reflog show HEAD

# Show reflog with dates
git reflog --date=iso

# Find lost commits
git reflog | grep "commit:"
git reflog | grep "checkout: moving"

# Recover from reflog
git checkout HEAD@{0}
git branch recovery-branch HEAD@{5}
\end{lstlisting}

\section{Undoing Changes}
\begin{lstlisting}[style=bash]
# Discard unstaged changes in a file
git checkout -- file.txt
git restore file.txt

# Discard all unstaged changes
git checkout -- .
git restore .

# Unstage a file (keep changes)
git reset HEAD file.txt

# Reset to previous commit (keep changes)
git reset --soft HEAD~1

# Reset to previous commit (discard changes)
git reset --hard HEAD~1

# Revert a specific commit
git revert commit-hash
\end{lstlisting}

\section{Tags}
\begin{lstlisting}[style=bash]
# Create tag
git tag v1.0.0

# Create annotated tag
git tag -a v1.0.0 -m "Release version 1.0.0"

# List tags
git tag -l

# Show tag details
git show v1.0.0

# Push tags
git push --tags

# Delete tag
git tag -d v1.0.0
git push origin --delete v1.0.0
\end{lstlisting}

\section{Workflow Example}
\begin{lstlisting}[style=bash]
# 1. Create and switch to new branch
git checkout -b feature/my-feature

# 2. Make changes and stage them
git add .

# 3. Commit changes
git commit -m "Add new feature"

# 4. Push to remote
git push -u origin feature/my-feature

# 5. After review, merge to main
git checkout main
git pull origin main
git merge feature/my-feature
git push origin main

# 6. Delete feature branch
git branch -d feature/my-feature
git push origin --delete feature/my-feature
\end{lstlisting}

\section{Checking Changes}
\begin{lstlisting}[style=bash]
# See what files changed in last commit
git show --stat HEAD

# See what changed since a commit
git diff abc123 --stat

# Check for untracked files
git status -u
\end{lstlisting}

\section{Cherry Picking}
\begin{lstlisting}[style=bash]
# Cherry pick a commit
git cherry-pick abc123

# Cherry pick without committing
git cherry-pick -n abc123

# Cherry pick a range of commits
git cherry-pick abc123..def456
\end{lstlisting}

\section{Advanced}
\begin{lstlisting}[style=bash]
# Bisect (find bug)
git bisect start
git bisect bad
git bisect good v1.0.0
git bisect reset

# Submodules
git submodule add url path
git submodule update --init

# Worktree
git worktree add ../branch-name branch-name
git worktree list

# Clean working directory
git clean -n  # dry run
git clean -f  # remove untracked
git clean -fd # remove directories
git clean -fdx # remove ignored files
\end{lstlisting}
